Field

The Field facade provides a library of reusable form field types used in Metabox, Page, User, and Taxonomy.

Available field types

Text inputs

Field::text('field_name')
Field::text('field_name', ['label' => 'Display Label', 'default' => 'default value'])

Field::email('contact_email')
Field::url('website_url', ['rules' => 'url'])
Field::number('price', ['rules' => 'numeric|min:0'])
Field::integer('quantity')
Field::password('api_secret')
Field::color('brand_color', ['default' => '#e84a4a'])

Textarea

Field::textarea('description', ['rows' => 5])
Field::editor('content')  // WordPress TinyMCE editor

Choice fields

Field::checkbox('is_featured', ['label' => 'Feature this post'])

Field::choice('status', [
    'choices' => ['Draft', 'Published', 'Archived'],
])

// Key-value choices
Field::choice('role', [
    'choices' => [
        'admin'  => 'Administrator',
        'editor' => 'Editor',
        'viewer' => 'Viewer',
    ],
    'expanded' => true,  // render as radio buttons
    'multiple' => false,
])

// Multiple select
Field::choice('tags', [
    'choices'  => ['PHP', 'JavaScript', 'CSS', 'HTML'],
    'multiple' => true,
])

Media

Field::media('featured_image')  // returns attachment ID
Field::media('pdf_upload', [
    'type' => 'application/pdf',
    'label' => 'Upload PDF',
])

Date & time

Field::date('event_date')
Field::date('event_date', ['format' => 'Y-m-d'])

Hidden

Field::hidden('_nonce_field', ['value' => wp_create_nonce('my-action')])

Common options

Field::text('name', [
    'label'       => 'Display Label',
    'placeholder' => 'Enter value here...',
    'default'     => 'Default value',
    'rules'       => 'required|min:3',
    'messages'    => ['required' => 'This field is required.'],
    'info'        => 'Helper text shown below the field.',
])

Validation rules

Field validation uses illuminate/validation rules. Common ones:

'rules' => 'required'
'rules' => 'required|min:3|max:100'
'rules' => 'email'
'rules' => 'url'
'rules' => 'numeric|min:0|max:999'
'rules' => 'integer'
'rules' => 'date'
'rules' => 'in:draft,published,archived'
