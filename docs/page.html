Page

The Page facade creates WordPress admin options pages — custom panels outside any post type.

Basic usage

use Meat\Support\Facades\Page;

Page::make('Theme Options', 'theme-options')->set();

Add settings fields

use Meat\Support\Facades\{Page, Field};

Page::make('Theme Options', 'theme-options')
    ->setPrefix('theme_')
    ->add(Field::text('tagline'))
    ->add(Field::media('logo'))
    ->add(Field::color('primary_color'))
    ->add(Field::checkbox('maintenance_mode'))
    ->set();

Organized sections (tabs)

use Meat\Support\Section;

Page::make('Theme Options', 'theme-options')
    ->add(new Section('general', 'General', [
        Field::text('tagline'),
        Field::media('logo'),
    ]))
    ->add(new Section('colors', 'Colors', [
        Field::color('primary'),
        Field::color('secondary'),
    ]))
    ->set();

Sub-pages

$parent = Page::make('MEAT Options', 'meat-options')->set();

Page::make('Advanced', 'meat-options-advanced')
    ->setParent($parent)
    ->setCapability('manage_options')
    ->add(Field::textarea('custom_css'))
    ->set();

Retrieve settings

// Fields are stored as WP options prefixed by setPrefix() value
$tagline = get_option('theme_tagline');
$logo    = get_option('theme_logo');
