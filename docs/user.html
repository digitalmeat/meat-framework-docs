User

The User facade adds custom meta fields to WordPress user edit screens.

Current user

use Meat\Support\Facades\User;

$user  = User::current();
$id    = User::current()->ID;
$email = User::current()->user_email;
$roles = User::current()->roles;
User::current()->has_cap('edit_posts');

Custom user meta fields

use Meat\Support\Facades\{User, Field};

User::meta()
    ->setPrefix('profile_')
    ->add(Field::text('phone'))
    ->add(Field::text('company'))
    ->add(Field::media('avatar_custom'))
    ->add(Field::choice('department', [
        'choices' => ['Engineering', 'Sales', 'Marketing', 'Support'],
    ]))
    ->set();

Grouped fields

use Meat\Support\Section;

User::meta()
    ->add(new Section('contact', 'Contact', [
        Field::text('phone'),
        Field::text('linkedin'),
    ]))
    ->add(new Section('profile', 'Profile', [
        Field::textarea('bio'),
        Field::media('avatar_custom'),
    ]))
    ->set();

Retrieve user meta

$phone = get_user_meta($user_id, 'profile_phone', true);
$bio   = get_user_meta($user_id, 'profile_bio',   true);

Validation

User::meta()
    ->add(Field::text('website', ['rules' => 'url']))
    ->add(Field::text('phone',   ['rules' => 'required|min:7']))
    ->set();
