Console (Artisan)

MEAT Framework uses Laravel Artisan. Run php artisan from project root to list all commands.

Built-in commands

# Setup
php artisan key:generate
php artisan salts:generate

# Scaffolding
php artisan make:controller PageController
php artisan make:model Book --migration
php artisan make:middleware CheckCapability
php artisan make:hook MyHook
php artisan make:command SendWeeklyDigest

# Database
php artisan migrate
php artisan migrate:rollback
php artisan migrate:status

# Cache
php artisan cache:clear
php artisan route:clear
php artisan view:clear

# WordPress
php artisan theme:install my-theme

# Dev
php artisan route:list
php artisan tinker

Custom commands

php artisan make:command SendWeeklyDigest
namespace App\Console\Commands;
use Illuminate\Console\Command;

class SendWeeklyDigest extends Command
{
    protected $signature   = 'digest:weekly {--dry-run}';
    protected $description = 'Send weekly digest to subscribers';

    public function handle(): int
    {
        if ($this->option('dry-run')) {
            $this->info('Dry run — no emails sent.');
            return Command::SUCCESS;
        }
        $users = get_users(['role' => 'subscriber']);
        $this->withProgressBar($users, fn($u) => wp_mail($u->user_email, 'Digest', 'Hi!'));
        $this->newLine();
        return Command::SUCCESS;
    }
}

Scheduled commands

// app/Console/Kernel.php
protected function schedule(Schedule $schedule): void
{
    $schedule->command('digest:weekly')->weekly()->mondays()->at('08:00');
    $schedule->command('cache:clear')->daily();
}
