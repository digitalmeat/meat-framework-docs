Introduction

MEAT Framework uses Laravel Blade (from illuminate/view) as its primary template engine. Blade files use the .blade.php extension and are compiled to plain PHP and cached — there is no runtime overhead.

Basic usage

Store views in your theme's views/ folder. Return them from routes or controllers:

// In a route
Route::any('front', function ($post, $query) {
 return view('home', ['posts' => $query->get_posts()]);
});

// In a controller
public function index()
{
 return view('pages.about');
}

Layouts & sections

<!-- views/layouts/main.blade.php -->
<!DOCTYPE html>
<html>
<head>
 @wp_head
</head>
<body>
 @include('partials.header')

 <main>@yield('content')</main>

 <aside>
 @section('sidebar')
 <p>Default sidebar</p>
 @show
 </aside>

 @include('partials.footer')
 @wp_footer
</body>
</html>
<!-- views/pages/home.blade.php -->
@extends('layouts.main')

@section('content')
 <h1>Welcome, {{ $name }}</h1>
 @foreach($posts as $post)
 <article>
 <h2>{{ $post->post_title }}</h2>
 </article>
 @endforeach
@endsection

@section('sidebar')
 @parent <!-- keeps parent content -->
 <p>Extra sidebar content</p>
@endsection

Control structures

<!-- If / elseif / else -->
@if(is_user_logged_in())
 <p>Welcome back!</p>
@elseif($pending)
 <p>Your account is pending.</p>
@else
 <p>Please log in.</p>
@endif

<!-- Unless (negated if) -->
@unless(is_user_logged_in())
 <a href="/login">Login</a>
@endunless

<!-- Foreach -->
@foreach($posts as $post)
 <li>{{ $post->post_title }}</li>
 <!-- $loop->first, $loop->last, $loop->index available -->
@endforeach

<!-- For -->
@for($i = 0; $i < 5; $i++)
 <span>{{ $i }}</span>
@endfor

<!-- Include -->
@include('partials.card', ['title' => 'My post'])

Components & slots

<!-- views/components/alert.blade.php -->
<div class="alert alert-{{ $type ?? 'info' }}">
 {{ $slot }}
</div>
@component('components.alert', ['type' => 'danger'])
 <strong>Error!</strong> Something went wrong.
@endcomponent

Echoing data

<!-- Escaped (safe) -->
{{ $name }}

<!-- Unescaped (use for trusted HTML, e.g. post content) -->
{!! $content !!}
{!! get_the_content() !!}

<!-- Null coalescing -->
{{ $title ?? 'Untitled' }}

<!-- Comments (not rendered in HTML) -->
{{-- This won't appear in source --}}

<!-- Raw curly braces (for JS frameworks like Vue/Alpine) -->
@{{ reactiveVar }}

WordPress-specific directives

<!-- WordPress loop -->
@loop
 <article>
 <h2>{{ Loop::title() }}</h2>
 {!! Loop::content() !!}
 <a href="{{ Loop::link() }}">Read more</a>
 </article>
@endloop

<!-- Custom query -->
@query(['post_type' => 'product', 'posts_per_page' => 6])
 <div class="product">
 <h3>{{ Loop::title() }}</h3>
 {{ Loop::thumbnail('medium') }}
 </div>
@endquery

<!-- Head & footer hooks -->
@wp_head <!-- outputs wp_head() -->
@wp_footer <!-- outputs wp_footer() -->

<!-- Dynamic template part -->
@template('parts.content', get_post_type())
<!-- Tries: parts.content-product, parts.content-page, …, then parts.content -->

Loop helper methods

The Loop class wraps WordPress loop functions for use inside @loop / @query:

Loop::id() // post ID
Loop::title() // post title
Loop::content() // post content (HTML)
Loop::excerpt() // post excerpt
Loop::link() // permalink
Loop::thumbnail('medium') // <img> tag
Loop::thumbnailUrl('medium') // image URL
Loop::author() // author display name
Loop::authorMeta('email') // author meta field
Loop::date('d/m/Y') // formatted date
Loop::category() // array of category terms
Loop::tags() // array of tag terms
Loop::terms('my-taxonomy') // array of custom taxonomy terms
Loop::postClass() // class="post-4 post type-post…"
Loop::nextPage() // next page link
Loop::previousPage() // previous page link
Loop::paginate() // pagination links

Extending Blade

Register custom directives from a service provider's boot() method using the Blade facade from illuminate/view:

use Illuminate\Support\Facades\Blade;

public function boot(): void
{
 Blade::directive('menu', function (string $expression) {
 return "<?php wp_nav_menu({$expression}); ?>";
 });

 Blade::directive('asset', function (string $expression) {
 return "<?php echo get_template_directory_uri() . '/dist/' . {$expression}; ?>";
 });
}
